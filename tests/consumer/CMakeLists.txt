cmake_minimum_required(VERSION 3.23)
project(FlagCVInstallConsumer LANGUAGES CXX)

find_package(FlagCV CONFIG REQUIRED)
add_executable(flagcv_install_consumer main.cpp)
target_link_libraries(flagcv_install_consumer PRIVATE FlagCV::flagcv)
option(FLAGCV_CONSUMER_NVIDIA "Exercise the installed NVIDIA backend" ON)
option(FLAGCV_CONSUMER_AMD "Exercise the installed AMD backend" OFF)
option(FLAGCV_CONSUMER_HYGON "Exercise the installed Hygon backend" OFF)
if(FLAGCV_CONSUMER_HIP)
  message(FATAL_ERROR "FLAGCV_CONSUMER_HIP was removed; use FLAGCV_CONSUMER_AMD or FLAGCV_CONSUMER_HYGON.")
endif()
if(FLAGCV_CONSUMER_AMD AND FLAGCV_CONSUMER_HYGON)
  message(FATAL_ERROR "Select only one HIP vendor and set hip_DIR to its SDK.")
endif()
if(FLAGCV_CONSUMER_AMD OR FLAGCV_CONSUMER_HYGON)
  find_package(hip CONFIG REQUIRED)
  if(FLAGCV_CONSUMER_AMD)
    set(vendor amd)
  else()
    set(vendor hygon)
  endif()
  get_target_property(hip_libraries hip::host INTERFACE_LINK_LIBRARIES)
  if((FLAGCV_CONSUMER_AMD AND NOT hip_libraries MATCHES "(^|::|/)amdhip64([.;>/]|$)") OR
     (FLAGCV_CONSUMER_HYGON AND NOT hip_libraries MATCHES "(^|::|/)galaxyhip([.;>/]|$)"))
    message(FATAL_ERROR "Select the ${vendor} SDK's hip::host with hip_DIR.")
  endif()
  target_compile_definitions(flagcv_install_consumer PRIVATE FLAGCV_CONSUMER_VENDOR="${vendor}")
  target_include_directories(flagcv_install_consumer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../cpp")
  target_link_libraries(flagcv_install_consumer PRIVATE hip::host)
elseif(FLAGCV_CONSUMER_NVIDIA)
  find_package(CUDAToolkit REQUIRED)
  target_compile_definitions(flagcv_install_consumer PRIVATE FLAGCV_CONSUMER_NVIDIA)
  target_link_libraries(flagcv_install_consumer PRIVATE CUDA::cudart)
endif()
